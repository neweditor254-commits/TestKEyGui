-- Optional helper for making HTTP requests from an executor environment.
-- Platoboost's official template may bring its own request function; use theirs if so.
local function httpRequest(options)
	local req = (syn and syn.request) or (http and http.request) or http_request or request
	if not req then
		return nil, "No HTTP request function available in this environment"
	end
	local ok, res = pcall(req, options)
	if not ok then return nil, tostring(res) end
	return res
end

-- Wrapper: runs the verifier safely and enforces a timeout.
local function verifyKeySafe(key)
	local result, finished = nil, false

	task.spawn(function()
		local ok, res = pcall(VerifyWithPlatoboost, key)
		if ok and type(res) == "table" and res.status then
			result = res
		elseif ok then
			result = { status = "error", message = "Malformed verifier response" }
		else
			result = { status = "error", message = tostring(res) }
		end
		finished = true
	end)

	local started = os.clock()
	while not finished do
		if os.clock() - started >= CONFIG.VerifyTimeout then
			return { status = "timeout" }
		end
		task.wait(0.05)
	end
	return result
end

local function makeButton(text, pos, size, color)
	local btn = new("TextButton", {
		Position = pos,
		Size = size,
		BackgroundColor3 = color,
		BorderSizePixel = 0,
		AutoButtonColor = false,
		Font = Enum.Font.GothamBold,
		Text = text,
		TextColor3 = Theme.Text,
		TextSize = 14,
		Parent = Content,
	}, { corner(10) })

	btn.MouseEnter:Connect(function()
		if btn:GetAttribute("Disabled") then return end
		tween(btn, EASE_FAST, { BackgroundColor3 = color:Lerp(Color3.new(1, 1, 1), 0.12) })
	end)
	btn.MouseLeave:Connect(function()
		tween(btn, EASE_FAST, { BackgroundColor3 = color })
	end)
	btn.MouseButton1Down:Connect(function()
		if btn:GetAttribute("Disabled") then return end
		tween(btn, TweenInfo.new(0.08), { Size = size - UDim2.fromOffset(4, 2) })
	end)
	btn.MouseButton1Up:Connect(function()
		tween(btn, EASE_FAST, { Size = size })
	end)
	return btn
end

local function notify(message, kind, duration)
	local color = (kind == "success" and Theme.Success)
		or (kind == "error" and Theme.Error)
		or (kind == "warning" and Theme.Warning)
		or Theme.AccentAlt

	local toast = new("Frame", {
		Size = UDim2.new(1, 0, 0, 44),
		BackgroundColor3 = Theme.Panel,
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		Parent = Toasts,
	}, { corner(10) })
	local tStroke = stroke(color, 1.5, 1)
	tStroke.Parent = toast

	local accent = new("Frame", {
		Size = UDim2.new(0, 4, 1, -16),
		Position = UDim2.new(0, 8, 0, 8),
		BackgroundColor3 = color,
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		Parent = toast,
	}, { corner(2) })

	local label = new("TextLabel", {
		Position = UDim2.new(0, 22, 0, 0),
		Size = UDim2.new(1, -30, 1, 0),
		BackgroundTransparency = 1,
		Font = Enum.Font.GothamMedium,
		Text = message,
		TextColor3 = Theme.Text,
		TextSize = 13,
		TextTransparency = 1,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextWrapped = true,
		Parent = toast,
	})

	tween(toast, EASE_SMOOTH, { BackgroundTransparency = 0.05 })
	tween(tStroke, EASE_SMOOTH, { Transparency = 0.3 })
	tween(accent, EASE_SMOOTH, { BackgroundTransparency = 0 })
	tween(label, EASE_SMOOTH, { TextTransparency = 0 })

	task.delay(duration or 3, function()
		if not toast.Parent then return end
		tween(toast, EASE_SMOOTH, { BackgroundTransparency = 1 })
		tween(tStroke, EASE_SMOOTH, { Transparency = 1 })
		tween(accent, EASE_SMOOTH, { BackgroundTransparency = 1 })
		local out = tween(label, EASE_SMOOTH, { TextTransparency = 1 })
		out.Completed:Wait()
		toast:Destroy()
	end)
end

local function setStatus(text, kind)
	local color = (kind == "success" and Theme.Success)
		or (kind == "error" and Theme.Error)
		or (kind == "warning" and Theme.Warning)
		or (kind == "loading" and Theme.AccentAlt)
		or Theme.SubText
	StatusLabel.Text = text
	tween(StatusLabel, EASE_FAST, { TextColor3 = color })
end

local function shake(guiObject)
	local origin = guiObject.Position
	for i = 1, 4 do
		local dir = (i % 2 == 0) and 1 or -1
		tween(guiObject, TweenInfo.new(0.05), {
			Position = origin + UDim2.fromOffset(dir * 6, 0),
		}).Completed:Wait()
	end
	tween(guiObject, TweenInfo.new(0.05), { Position = origin })
end

	if isLoading then
		Spinner.Visible = true
		StatusLabel.Position = UDim2.new(0, 28, 0, 0)

	end
end

local isVerifying = false
local isFinished  = false  -- set once success flow begins (prevents double init)

local StartMainScript -- forward declaration

local function showSuccessAndContinue()
	isFinished = true
	SuccessFrame.Visible = true
	tween(SuccessFrame, EASE_SMOOTH, { BackgroundTransparency = 0 })
	tween(Content, EASE_SMOOTH, {}) -- (content is covered by SuccessFrame)

	tween(SuccessIcon, EASE_BACK, { Size = UDim2.fromOffset(72, 72), TextTransparency = 0 })
	task.delay(0.15, function() tween(SuccessText, EASE_SMOOTH, { TextTransparency = 0 }) end)
	task.delay(0.3,  function() tween(SuccessSub,  EASE_SMOOTH, { TextTransparency = 0 }) end)

	notify("Key Verified!", "success", 2.5)

	task.wait(CONFIG.SuccessDuration)

	-- Fade out and destroy the key GUI
	tween(Main, EASE_SMOOTH, { Size = UDim2.new(0.9, 0, 0, 0), BackgroundTransparency = 1 })
	task.wait(0.4)
	setLoading(false)
	ScreenGui:Destroy()

	StartMainScript()
end

local function onVerifyPressed()
	if isVerifying or isFinished then return end

	local key = (KeyBox.Text or ""):gsub("%s+", "")

	-- Empty / malformed key
	if key == "" then
		setStatus("Please enter a key", "warning")
		notify("Key field is empty", "warning")
		task.spawn(shake, KeyBox)
		return
	end
	if #key < CONFIG.MinKeyLength then
		setStatus("Invalid or expired key", "error")
		notify("Invalid or expired key", "error")
		task.spawn(shake, KeyBox)
		return
	end

	-- Lock UI
	isVerifying = true
	setButtonDisabled(VerifyBtn, true, "Verifying...")
	setButtonDisabled(GetKeyBtn, true)
	KeyBox.TextEditable = false
	setLoading(true)
	setStatus("Verifying key...", "loading")

	local result = verifyKeySafe(key)

	-- Unlock UI
	setLoading(false)
	isVerifying = false
	KeyBox.TextEditable = true
	setButtonDisabled(VerifyBtn, false, "Verify")
	setButtonDisabled(GetKeyBtn, false)

	local status = result.status
	if status == "valid" then
		setStatus("Key accepted", "success")
		task.spawn(showSuccessAndContinue)

	elseif status == "invalid" then
		setStatus("Invalid or expired key", "error")
		notify("Invalid or expired key", "error")
		task.spawn(shake, KeyBox)

	elseif status == "expired" then
		setStatus("Invalid or expired key", "error")
		notify("Invalid or expired key", "error")
		task.spawn(shake, KeyBox)

	elseif status == "timeout" then
		setStatus("Verification timed out. Try again.", "warning")
		notify("Verification timed out", "warning")

	else -- "error" or anything unexpected
		setStatus("Network/API error. Try again later.", "error")
		notify(result.message or "Network/API error", "error", 4)
		warn("[XreztHub] Verification error:", result.message)
	end
end
